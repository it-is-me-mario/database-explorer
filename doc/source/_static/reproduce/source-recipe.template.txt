# %% Gross output: X for IOT; Xc for SUT

# %% Domestic use: only matching origin and destination regions
z = region_table(db.query("U" if sut else "Z"), regions)

y_total = region_table(y, regions)

    "Intermediate": np.diag(z.to_numpy()) * scale,
    "Final": np.diag(y_total.to_numpy()) * scale,
}, index=regions)
domestic_use["Total"] = domestic_use["Intermediate"] + domestic_use["Final"]
domestic_use.to_csv(OUTPUT / "domestic_use.csv")
